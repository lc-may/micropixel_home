cmake_minimum_required(VERSION 3.20)
project(micropixel_rv32 LANGUAGES C ASM)
get_filename_component(REPOSITORY_ROOT "../../.." ABSOLUTE)
set(WAMR_ROOT_DIR "${REPOSITORY_ROOT}/firmware/espressif/components/wasm-micro-runtime")
set(WAMR_BUILD_PLATFORM "micropixel-rv32")
set(WAMR_BUILD_TARGET "RISCV32_ILP32F")
set(SHARED_PLATFORM_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/platform.cmake")
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_INTERP 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_BUILD_SHARED_MEMORY 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_DISABLE_WAKEUP_BLOCKING_OP 1)
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
add_executable(micropixel-rv32.elf ${WAMR_RUNTIME_LIB_SOURCE} startup.S runner.c)
target_include_directories(micropixel-rv32.elf PRIVATE "${REPOSITORY_ROOT}/guest")
target_compile_options(micropixel-rv32.elf PRIVATE -Os -ffunction-sections -fdata-sections -fno-stack-protector)
target_link_options(micropixel-rv32.elf PRIVATE -nostartfiles "-T${CMAKE_CURRENT_SOURCE_DIR}/linker.ld"
    -Wl,--gc-sections "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/runner.map")
target_link_libraries(micropixel-rv32.elf PRIVATE m c gcc)
set_property(TARGET micropixel-rv32.elf APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/linker.ld")
