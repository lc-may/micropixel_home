cmake_minimum_required(VERSION 3.20)
# FreeRTOSConfig.h must be visible while the SDK creates its kernel target.
include_directories("${CMAKE_CURRENT_LIST_DIR}/../../config")
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/sdk.cmake")
if(CONFIG_DISPLAY_DEFAULT AND CONFIG_LVGL9_PORT_DISP)
    sdk_add_include_directories("${CMAKE_CURRENT_LIST_DIR}")
    if(MPX_DISPLAY_GPIO_SOURCE)
        target_sources(app PRIVATE "${MPX_DISPLAY_GPIO_SOURCE}")
    elseif(CONFIG_DISPLAY_DEFAULT_TYPE_GRAM OR NOT CONFIG_DISPLAY_DEFAULT_DSI)
        message(FATAL_ERROR "Set DISPLAY_GPIO_SOURCE to the panel's Orion pin-mux implementation")
    endif()
    set(_display_source "${MPX_ROOT}/examples/display_smoke.c")
    if(MPX_NATIVE_VIDEO_TEST)
        set_source_files_properties("${_display_source}" PROPERTIES COMPILE_DEFINITIONS MPX_NATIVE_VIDEO_TEST=1)
        sdk_set_main_file("${_display_source}")
    else()
        target_sources(app PRIVATE "${_display_source}")
    endif()
endif()
if(MPX_NATIVE_VIDEO_TEST)
    project(micropixel_orion_display)
else()
    if(MPX_HOST_PORT)
        include("${MPX_ROOT}/boards/bl618dg/orion/host.cmake")
    endif()
    if(MPX_HOST_PORT OR MPX_AUDIO_TEST)
        include("${MPX_ROOT}/boards/bl618dg/orion/audio.cmake")
    endif()
    include("${MPX_ROOT}/cmake/runtime.cmake")
    if(MPX_AUDIO_TEST)
        project(micropixel_orion_audio)
    elseif(MPX_HOST_PORT)
        project(micropixel_orion_host)
    else()
        project(micropixel_bl618dg)
    endif()
endif()
